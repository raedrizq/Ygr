  <!-- GAME SCRIPT LOGIC -->
  <script>
    const AVATARS = ['🐶', '🐱', '🦊', '🦁', '🐸', '🐼', '🐯', '🦄', '🐙', '🐵', '🦉', '👑'];
    const QUESTIONS = [
      { id: 1, category: "حيوانات", q: "ما هو الحيوان المعروف بسفينة الصحراء؟", correct: "الجمل", options: ["الجمل", "الفهد", "السمك", "النعامة"] },
      { id: 2, category: "علوم", q: "ما هو الغاز الأساسي لتنفس الإنسان؟", correct: "الأكسجين", options: ["الأكسجين", "النيتروجين", "الخشب", "البنزين"] },
      { id: 3, category: "جغرافيا", q: "ما هي عاصمة جمهورية مصر العربية؟", correct: "القاهرة", options: ["القاهرة", "لندن", "طوكيو", "الرياض"] },
      { id: 4, category: "معلومات عامة", q: "كم عدد أيام الأسبوع؟", correct: "7 أيام", options: ["7 أيام", "30 يوم", "3 أيام", "100 يوم"] },
      { id: 5, category: "طعام", q: "فاكهة حمراء وبها بذور سوداء وتؤكل في الصيف؟", correct: "البطيخ", options: ["البطيخ", "الخيار", "اللحم", "الخبز"] },
      { id: 6, category: "رياضة", q: "كم عدد لاعبي فريق كرة القدم داخل الملعب؟", correct: "11 لاعب", options: ["11 لاعب", "50 لاعب", "لاعب واحد", "2 لاعبان"] },
      { id: 7, category: "تكنولوجيا", q: "ما الجهاز الذي تستخدمه للاتصال والتصفح؟", correct: "الهاتف الذكي", options: ["الهاتف الذكي", "الغسالة", "الثلاجة", "المروحة"] },
      { id: 8, category: "تاريخ", q: "من هم بناة أهرامات الجيزة الخالدة؟", correct: "الفراعنة", options: ["الفراعنة", "الرومان", "فضائيون", "الرواد"] },
      { id: 9, category: "مضحك", q: "ما الذي ينزل من السحاب عندما تمطر السماء؟", correct: "المطر / الماء", options: ["المطر / الماء", "عصير برتقال", "بيتزا طازجة", "نقود ورقية"] },
      { id: 10, category: "سهل جدًا", q: "ما هي نتيجة 1 + 1؟", correct: "2", options: ["2", "100", "صفر", "مليون"] }
    ];

    let myAvatar = '🦊';
    let myName = 'لاعب';
    let peer = null;
    let conn = null;
    let isHost = false;
    let roomCode = '';
    
    let gameState = {
      round: 0,
      scores: {},
      players: {}
    };

    // Initialize UI
    window.onload = () => {
      lucide.createIcons();
      renderAvatars();

      // Check URL for room parameter
      const urlParams = new URLSearchParams(window.location.search);
      const roomParam = urlParams.get('room');
      if (roomParam) {
        document.getElementById('input-room-code').value = roomParam;
      }
    };

    function renderAvatars() {
      const container = document.getElementById('avatar-selector');
      container.innerHTML = AVATARS.map(av => `
        <button onclick="selectAvatar('${av}')" class="avatar-btn text-xl p-2 rounded-xl bg-slate-900 border border-slate-700 transition ${av === myAvatar ? 'border-amber-400 bg-purple-900/40' : ''}">
          ${av}
        </button>
      `).join('');
    }

    function selectAvatar(av) {
      myAvatar = av;
      document.getElementById('my-avatar').innerText = av;
      renderAvatars();
    }

    function showScreen(id) {
      ['screen-home', 'screen-lobby', 'screen-game', 'screen-results'].forEach(s => {
        document.getElementById(s).classList.add('hidden');
      });
      document.getElementById(id).classList.remove('hidden');
    }

    // PeerJS Networking setup
    function createRoom() {
      myName = document.getElementById('input-name').value || 'لاعب 1';
      document.getElementById('my-name').innerText = myName;

      roomCode = 'JAWB-' + Math.floor(1000 + Math.random() * 9000);
      isHost = true;

      peer = new Peer(roomCode);

      peer.on('open', (id) => {
        document.getElementById('display-room-code').innerText = id;
        document.getElementById('chat-container').classList.remove('hidden');
        showScreen('screen-lobby');

        gameState.players[id] = { name: myName, avatar: myAvatar, score: 0 };
        renderLobbyPlayers();
      });

      peer.on('connection', (connection) => {
        conn = connection;
        setupConnectionListeners();
      });
    }

    function joinRoom() {
      myName = document.getElementById('input-name').value || 'لاعب 2';
      document.getElementById('my-name').innerText = myName;

      roomCode = document.getElementById('input-room-code').value.trim().toUpperCase();
      if (!roomCode) return alert('يرجى أدخال كود الغرفة أولاً');

      isHost = false;
      peer = new Peer();

      peer.on('open', () => {
        conn = peer.connect(roomCode);
        setupConnectionListeners();
      });
    }

    function setupConnectionListeners() {
      conn.on('open', () => {
        document.getElementById('chat-container').classList.remove('hidden');
        showScreen('screen-lobby');

        // Send info to peer
        conn.send({
          type: 'JOIN',
          name: myName,
          avatar: myAvatar
        });
      });

      conn.on('data', (data) => {
        handleIncomingData(data);
      });
    }

    function handleIncomingData(data) {
      switch (data.type) {
        case 'JOIN':
          gameState.players[conn.peer] = { name: data.name, avatar: data.avatar, score: 0 };
          renderLobbyPlayers();
          if (isHost) {
            conn.send({ type: 'SYNC_LOBBY', players: gameState.players });
          }
          break;

        case 'SYNC_LOBBY':
          gameState.players = data.players;
          renderLobbyPlayers();
          break;

        case 'START_GAME':
          gameState.round = 0;
          showScreen('screen-game');
          loadRoundQuestion();
          break;

        case 'ANSWER':
          if (isHost) {
            // Calculate score & move next
            gameState.players[conn.peer].score += data.points;
            gameState.round++;
            
            if (gameState.round < QUESTIONS.length) {
              broadcast({ type: 'NEXT_ROUND', round: gameState.round, players: gameState.players });
              loadRoundQuestion();
            } else {
              broadcast({ type: 'GAME_OVER', players: gameState.players });
              renderResults();
            }
          }
          break;

        case 'NEXT_ROUND':
          gameState.round = data.round;
          gameState.players = data.players;
          loadRoundQuestion();
          break;

        case 'GAME_OVER':
          gameState.players = data.players;
          renderResults();
          break;

        case 'CHAT':
          appendChatMessage(data.sender, data.text);
          break;

        case 'TAUNT':
          alert(`${data.sender}: ${data.text}`);
          break;
      }
    }

    function broadcast(data) {
      if (conn) conn.send(data);
      handleIncomingData(data);
    }

    function renderLobbyPlayers() {
      const container = document.getElementById('players-list');
      container.innerHTML = Object.values(gameState.players).map(p => `
        <div class="flex items-center gap-2 p-2.5 bg-slate-900 rounded-xl border border-slate-700">
          <span class="text-2xl">${p.avatar}</span>
          <div class="text-right">
            <div class="text-xs font-bold text-white">${p.name}</div>
            <span class="text-[10px] text-emerald-400">متصل ⚡</span>
          </div>
        </div>
      `).join('');
    }

    function startGame() {
      broadcast({ type: 'START_GAME' });
    }

    function loadRoundQuestion() {
      const q = QUESTIONS[gameState.round];
      document.getElementById('question-category').innerText = q.category;
      document.getElementById('question-text').innerText = q.q;

      // Render Scores Header
      document.getElementById('game-scores-bar').innerHTML = Object.values(gameState.players).map(p => `
        <span class="bg-slate-900 px-3 py-1 rounded-lg text-amber-300 font-bold">
          ${p.avatar} ${p.name}: ${p.score} نقطة
        </span>
      `).join('');

      // Options
      const container = document.getElementById('options-container');
      container.innerHTML = q.options.map((opt) => `
        <button onclick="submitAnswer('${opt}')" class="p-3.5 rounded-xl font-bold text-sm text-right bg-slate-900 hover:bg-purple-900/40 border border-slate-700 text-slate-200 transition">
          ${opt}
        </button>
      `).join('');
    }

    function submitAnswer(chosenOption) {
      const q = QUESTIONS[gameState.round];
      const isWrongAnswer = chosenOption !== q.correct;
      const points = isWrongAnswer ? 1 : 0; // Wrong answer gives 1 point!

      if (isHost) {
        gameState.players[peer.id].score += points;
        gameState.round++;
        if (gameState.round < QUESTIONS.length) {
          broadcast({ type: 'NEXT_ROUND', round: gameState.round, players: gameState.players });
        } else {
          broadcast({ type: 'GAME_OVER', players: gameState.players });
        }
      } else {
        conn.send({ type: 'ANSWER', points: points });
      }
    }

    function renderResults() {
      showScreen('screen-results');
      const list = document.getElementById('results-list');
      const sorted = Object.values(gameState.players).sort((a,b) => b.score - a.score);
      
      list.innerHTML = sorted.map((p, idx) => `
        <div class="flex justify-between items-center p-3 bg-slate-900 rounded-xl border border-slate-700">
          <div class="flex items-center gap-2">
            <span class="font-bold text-amber-400">#${idx + 1}</span>
            <span>${p.avatar}</span>
            <span class="font-bold text-sm">${p.name}</span>
          </div>
          <span class="font-black text-amber-400">${p.score} نقطة</span>
        </div>
      `).join('');
    }

    function copyInviteLink() {
      navigator.clipboard.writeText(roomCode);
      document.getElementById('copy-btn-text').innerText = 'تم النسخ!';
      setTimeout(() => document.getElementById('copy-btn-text').innerText = 'نسخ الكود', 2000);
    }

    // Chat Logic
    function toggleChat() {
      document.getElementById('chat-box').classList.toggle('hidden');
    }

    function sendChatMessage(e) {
      e.preventDefault();
      const input = document.getElementById('chat-input');
      const text = input.value.trim();
      if (!text) return;

      broadcast({ type: 'CHAT', sender: myName, text: text });
      input.value = '';
    }

    function appendChatMessage(sender, text) {
      const container = document.getElementById('chat-messages');
      container.innerHTML += `
        <div class="bg-slate-800 p-2 rounded-xl border border-slate-700">
          <span class="text-purple-400 font-bold">${sender}: </span>
          <span class="text-slate-200">${text}</span>
        </div>
      `;
      container.scrollTop = container.scrollHeight;
    }

    function sendQuickTaunt(msg) {
      broadcast({ type: 'TAUNT', sender: myName, text: msg });
    }

    function restartGame() {
      showScreen('screen-home');
      location.reload();
    }
  </script>
